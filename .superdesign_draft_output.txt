<script>
  // 1. Mercury Button Elastic Morphing
  document.querySelectorAll('.mercury-btn').forEach(btn => {
    btn.addEventListener('mousemove', (e) => {
      const rect = btn.getBoundingClientRect();
      const x = e.clientX - rect.left - rect.width / 2;
      const y = e.clientY - rect.top - rect.height / 2;
      
      // Stretch towards cursor
      btn.style.transform = `translate(${x * 0.3}px, ${y * 0.3}px) scale(1.05)`;
      
      // Morph edges slightly
      const xPerc = (e.clientX - rect.left) / rect.width * 100;
      const yPerc = (e.clientY - rect.top) / rect.height * 100;
      btn.style.borderRadius = `${xPerc}% ${100-xPerc}% ${yPerc}% ${100-yPerc}% / ${yPerc}% ${100-yPerc}% ${xPerc}% ${100-xPerc}%`;
    });
    btn.addEventListener('mouseleave', () => {
      btn.style.transform = 'translate(0, 0) scale(1)';
      btn.style.borderRadius = '9999px';
    });
  });

  // 2. Focus Lens Effect
  const lensContainer = document.getElementById('focus-lens-section');
  if (lensContainer) {
    lensContainer.addEventListener('mousemove', (e) => {
      const rect = lensContainer.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;
      lensContainer.style.setProperty('--lens-x', `${x}px`);
      lensContainer.style.setProperty('--lens-y', `${y}px`);
    });
  }

  // 3. 3D Depth Stack Scroll Logic
  const stackContainer = document.querySelector('.depth-stack-viewport');
  const cards = document.querySelectorAll('.depth-card');
  
  window.addEventListener('scroll', () => {
    const rect = stackContainer.getBoundingClientRect();
    const scrollProgress = Math.max(0, Math.min(1, -rect.top / (rect.height - window.innerHeight)));
    
    cards.forEach((card, index) => {
      const initialZ = parseInt(card.dataset.z);
      const zProgress = scrollProgress * 2500; // Total Z movement
      const currentZ = initialZ + zProgress;
      
      // Fly toward camera and fade out
      if (currentZ > 400) {
        card.style.opacity = Math.max(0, 1 - (currentZ - 400) / 400);
      } else {
        card.style.opacity = 1;
      }
      
      card.style.transform = `translateZ(${currentZ}px) rotateX(${currentZ * 0.05}deg)`;

      // Prevent interaction if card is "behind" camera
      card.style.pointerEvents = currentZ > 800 ? 'none' : 'auto';
    });
  });

  // 4. Card 3D Tilt Hover (when pinned/near)
  cards.forEach(card => {
    card.addEventListener('mousemove', (e) => {
      const rect = card.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;
      const centerX = rect.width / 2;
      const centerY = rect.height / 2;
      const rotateX = (y - centerY) / 10;
      const rotateY = (centerX - x) / 10;

      const baseTransform = card.style.transform.split(' rotateX')[0];
      card.style.transform = `${baseTransform} rotateX(${rotateX}deg) rotateY(${rotateY}deg)`;
      card.classList.add('pinned');
    });
    card.addEventListener('mouseleave', () => {
      card.classList.remove('pinned');
    });
  });

  // Intersection Observer for Reveal
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) entry.target.classList.add('active');
    });
  }, { threshold: 0.1 });
  document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

  // Nav Scroll Effect
  const nav = document.getElementById('main-nav');
  window.addEventListener('scroll', () => {
    if (window.scrollY > 100) nav.classList.add('nav-scrolled');
    else nav.classList.remove('nav-scrolled');
  });
</script>
